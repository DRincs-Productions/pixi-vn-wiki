詳しくは<DynamicLink href="/start/translate">こちら</DynamicLink>で説明していますが、Pixi’VN では [i18next](https://www.i18next.com/) などのライブラリを使ってゲーム内のテキストを翻訳できます。また、***ink* と Pixi’VN の連携**でも、ライブラリを使って ***ink*** のテキストを翻訳できます。

Pixi’VN では、[`onInkTranslate`](/jsdoc/pixi-vn-ink/index/functions/onInkTranslate) 関数を使って翻訳イベントを捕捉できます。

```ts tab="lib/hooks/ink-hooks.tsx"
import { onInkTranslate } from "@drincs/pixi-vn-ink";
import { useEffect } from "react";
import { useTranslation } from "react-i18next";

export default function useInkInitialization() {
    const { t } = useTranslation(["narration"]);
    useEffect(() => {
        onInkTranslate(t);
    }, [t]);

    return null;
}
```

```ink tab="ink/start.ink"
=== start ===
Hello, my name is [joe]
-> DONE
```

```json tab="locales/es.json"
{
    "narration": {
        "Hello, my name is [joe]": "Hola, mi nombre es {{joe}}"
        // ...
    }
}
```

## 翻訳ファイルの自動生成

<Callout title="テンプレート" type="info">

すべてのテンプレートで、設定にある次のボタンを使うと、翻訳に使用する JSON ファイルを自動生成できます。
![dowload-translate](/dowload-translate.png)

</Callout>

翻訳ファイルを簡単に作成できるよう、[`generateJsonInkTranslation`](/jsdoc/pixi-vn-ink/index/functions/generateJsonInkTranslation) 関数を使って ink テキストから JSON 構造を自動生成できます。

```ts title="lib/i18n.ts"
import { generateJsonInkTranslation } from "@drincs/pixi-vn-ink";
import i18n from "i18next";
import { convertInkToJson } from "./utils/ink-utility";

function getLocalesResource(
    lng: string,
): Promise<Record<string, Record<string, string>>> {
    return import(`./../locales/${lng}.json`);
}

async function generateResourceToTranslate(lng: string): Promise<any> {
    let res = await getLocalesResource(lng);
    res = { ...res };
    if (!res) {
        res = {};
    }
    if (!res.narration) {
        res.narration = {};
    }
    if (res.default) {
        delete res.default;
    }
    const manifest = await import("@/assets/ink-manifest.gen.json");
    for (const path of manifest.default) {
        const element = await fetch(path).then((r) => r.json());
        element && (await generateJsonInkTranslation(element, res.narration));
    }
    return res;
}

export async function downloadResourceToTranslate() {
    const lng = i18n.options.fallbackLng?.toString() || "en";
    const data = await generateResourceToTranslate(lng);
    const jsonString = JSON.stringify(data);
    // download the save data as a JSON file
    const blob = new Blob([jsonString], { type: "application/json" });
    // download the file
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = `strings_${lng}.json`;
    a.click();
}
```
